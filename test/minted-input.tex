\newcommand{\checkinputlanguage}[1]{%
  \edef\actualinputlanguage{\languagename}%
  \ifdefstring{\actualinputlanguage}{#1}{}{%
    \PackageError{minted-input-test}{Expected #1, got \languagename}{}%
  }%
}
