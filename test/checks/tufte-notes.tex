\newcommand\checknotedirection{%
  \ifLuaTeX
    \ifnum\bbl@thepardir=1\relax\else
      \errmessage{RTL note paragraph direction lost}%
    \fi
  \fi
}
\ifXeTeX
  % The optional-argument wrapper calls the internal control sequence whose
  % name begins with the current escape character. Check its patched body.
  \AtBeginDocument{%
    \ifnum\escapechar=92\relax\else
      \errmessage{Unexpected escapechar in Tufte patch test}%
    \fi
    \@for\bbdtestnote:=LTR,RTL\do{%
      \expandafter\let\expandafter\bbdtestbody
        \csname\expandafter\string\csname @\bbdtestnote footnotetext\endcsname\endcsname
      \patchcmd{\bbdtestbody}{\bbd@tufte@anchor}{}{}%
        {\errmessage{Missing Tufte note anchor in patched XeTeX body}}%
      \patchcmd{\bbdtestbody}{\bbd@tufte@restore@anchor}{}{}%
        {\errmessage{Missing Tufte anchor restore in patched XeTeX body}}%
    }%
    % Each paragraph needs its own RTL run, not only the first one.
    \let\bbdtestreset\@RTLbidituftesidenotemarginparreset
    \patchcmd{\bbdtestreset}{\everypar{\@minipagefalse\beginR}}{}{}%
      {\errmessage{XeTeX RTL note paragraph reset is not patched}}%
  }
\fi
\makeatother
