\ifLuaTeX
  \def\CheckDir#1{%
    \ifnum\bbl@thetextdir=#1\else\errmessage{Babel text direction mismatch}\fi
    \ifnum\textdirection=#1\else\errmessage{Lua text direction mismatch}\fi
    \ifnum\bbl@thedir=\numexpr\bbl@thepardir*4+#1\relax\else
      \errmessage{Babel combined direction mismatch}\fi
    \ifnum\bbl@attr@dir=\bbl@thedir\relax\else
      \errmessage{Babel direction attribute mismatch}\fi}
  \def\CheckExplicit#1{%
    \ifcsname bbd@lua@explicit@direction\endcsname
      \ifnum#1=1\else\errmessage{Unexpected explicit direction}\fi
      \ifx\bbd@lua@explicit@direction\@empty\else
        \errmessage{Explicit direction marker is not empty}\fi
    \else
      \ifnum#1=0\else\errmessage{Missing explicit direction}\fi
    \fi}
\else
  \def\CheckDir#1{}
  \def\CheckExplicit#1{}
\fi
\makeatother
