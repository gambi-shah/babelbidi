% !TeX program = lualatex
% Set language roles with package options; \lr, \rl and latin select
% the matching language and font, while leaving the surrounding text intact.
% The table shows the automatically selected roles for inspection.
\documentclass{article}

\usepackage[french,english,persian,greek,rtl=arabic]{babelbidi}

\babelfont{rm}[arabic]{Amiri}
\babelfont{rm}[persian]{Vazirmatn}
\babelfont{sf}{Latin Modern Sans}
\babelfont{tt}{Latin Modern Mono}

\begin{document}

\section{Language Mappings}

\makeatletter
\begin{tabular}{ll}
    Code  & \texttt{\bbd@main@code}  \\
    Latin & \texttt{\bbd@main@latin} \\
    LTR   & \texttt{\bbd@main@ltr}   \\
    RTL   & \texttt{\bbd@main@rtl}
\end{tabular}
\makeatother

\section{Bidirectional Text}

\noindent
\lr{Left-to-right text}

\rl{متن راست‌به‌چپ}

\begin{latin}
    This is a Latin paragraph.
\end{latin}

\end{document}
