% Test babelbidi-api-core.def; compile with lualatex, xelatex or pdflatex.
\documentclass{article}

\usepackage{babelbidi}

\makeatletter
\newcommand{\AssertRTL}{%
    \if@RTL\else
    \PackageError{babelbidi-api-core}{Expected RTL direction}{}%
    \fi
}
\newcommand{\AssertLTR}{%
    \if@RTL
    \PackageError{babelbidi-api-core}{Expected LTR direction}{}%
    \fi
}
\makeatother

\begin{document}
    \section{Direction commands and environments}
    Default LTR text.\AssertLTR

    \setRTL\AssertRTL
    RTL paragraph with English text.\par
    \setLTR\AssertLTR
    LTR paragraph.\par

    {\setRL\AssertRTL RTL text in a group.\par}
    \AssertLTR
    {\unsetLTR\AssertRTL RTL text after unsetLTR.\par}
    \AssertLTR
    {\unsetLR\AssertRTL RTL text after unsetLR.\par}
    \AssertLTR
    {\setRTL\AssertRTL\unsetRL\AssertLTR LTR text after unsetRL.\par}
    \AssertLTR
    \setLR\AssertLTR
    \unsetRTL\AssertLTR

    \begin{RTL}
        \AssertRTL Text in an RTL environment.
        \begin{LTR}
            \AssertLTR Text in a nested LTR environment.
        \end{LTR}
        \AssertRTL
    \end{RTL}
    \AssertLTR

    \begin{LTR}
        \AssertLTR Text in the LTR environment.
    \end{LTR}

    \leavevmode\beginR Text in a right-to-left group.\endR
    \beginL Text in a left-to-right group.\endL

    \section{Inline direction and numerals}
    \LRE{LTR text} \LR{LTR alias}.
    \RLE{RTL-directed English text} \RL{RTL-directed English text}.
    Numeral: \localnumeral{123}; section counter: \localnumeral*{section}.

    \section{Directional boxes}
    \fbox{\hboxL{Left-to-right box}}
    \fbox{\hboxR{Right-to-left box}}

    \medskip\noindent
    \fbox{\hboxL to 4cm{Left\hfil Right}}
    \fbox{\hboxR spread 1cm{Right\hfil Left}}

    \medskip\noindent
    \fbox{\vboxL to 2cm{\hsize=3cm First line\par\vfil Last line\par}}
    \fbox{\vtopR spread 5mm{\hsize=3cm
            First line\par\vfil Last line\par}}

    \medskip\noindent
    \fbox{\vboxR{\hsize=3cm First line\par Second line\par}}
    \fbox{\vtopL{\hsize=3cm First line\par Second line\par}}

    \medskip
    \setbox0=\hboxL to 5cm{Left\hfil Right}
    \fbox{\box0}
    \setbox1=\vtopR{\hsize=3cm First line\par Second line\par}
    \fbox{\box1}

    \clearpage
    \section{Directed footnotes and rules}
    \setlength{\footnoterulewidth}{.4\columnwidth}
    \leftfootnoterule
    Left rule.\LTRfootnote[3]{LTR footnote text.\par Second paragraph.}
    RTL note.\RTLfootnote[4]{RTL-directed English footnote text.}

    \clearpage
    \rightfootnoterule
    Right rule.\RTLfootnote{RTL-directed English footnote text.}

    \clearpage
    \textwidthfootnoterule
    Full-width rule.\LTRfootnote{LTR footnote text.}

    \clearpage
    \autofootnoterule
    Automatic rule, RTL first.\RTLfootnote{RTL-directed English footnote text.}
    Second note.\LTRfootnote{LTR footnote text.}

    \clearpage
    Automatic rule, LTR first.\LTRfootnote{LTR footnote text.}
\end{document}
