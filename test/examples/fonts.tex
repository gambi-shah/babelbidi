% Unicode-engine example: select main, Latin and custom font families.
\documentclass{article}

\usepackage{babelbidi}

\babelprovide[import,main]{persian}
\babelprovide[import]{english}

\settextfont{Vazirmatn}
\setlatintextfont{TeX Gyre Termes}
\setlatinsansfont{TeX Gyre Heros}
\setlatinmonofont{DejaVu Sans Mono}

% Define a new command name; \latinfont itself already belongs to BabelBidi.
\deflatinfont{\customlatinfont}{TeX Gyre Termes}

\begin{document}

    متن فارسی با فونت اصلی.

    \selectlanguage{english}

    English text in roman font.

    \textsf{English text in sans-serif font.}

    \texttt{English text in Latin monospace font.}

    {\latinfont The selected Latin font family.}

    {\customlatinfont A separately defined Latin font family.}

    \selectlanguage{persian}

    فارسی و
    {\latinttfamily Latin monospace}
    در یک متن.

\end{document}
