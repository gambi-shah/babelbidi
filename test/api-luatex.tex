\babelprovide[main,import]{english}
\babelprovide[import,onchar=fonts]{persian}

\babelfont[english]{rm}{Times New Roman}
\babelfont[persian]{rm}{HM XNiloofar}

\makeatletter
\newcommand{\AssertRTL}{%
    \if@RTL\else
    \PackageError{api-luatex}{Expected RTL direction}{}
    \fi
}
\newcommand{\AssertLTR}{%
    \if@RTL
    \PackageError{api-luatex}{Expected LTR direction}{}
    \fi
}
\makeatother

\begin{document}

    \section*{Direction}

    Default LTR text.
    \AssertLTR

    \setRTL
    \AssertRTL
    متن راست‌به‌چپ.

    \setLTR
    \AssertLTR
    LTR paragraph.

    \setRL
    \AssertRTL
    متن فارسی.

    \unsetLTR
    \AssertRTL

    \setLR
    \AssertLTR
    LTR paragraph.

    \unsetRTL
    \AssertLTR

    \begin{RTL}
        \AssertRTL
        متن فارسی در محیط RTL.
    \end{RTL}
    \AssertLTR

    \begin{LTR}
        \AssertLTR
        Text in the LTR environment.
    \end{LTR}
    \AssertLTR

    \beginR RTL group. \endR
    \beginL LTR group. \endL

    \section*{Inline Direction}

    \LRE{LTR text} \LR{LTR text}

    \RLE{متن فارسی} \RL{متن فارسی}

    \section*{Directional Boxes}

    \fbox{\hboxL{English — فارسی}}
    \fbox{\hboxR{English — فارسی}}

    \medskip

    \fbox{\vboxL to 2cm{\hsize=3cm
            First line\par \vfil Last line\par}}

            سطر نخست\par \vfil سطر آخر\par}}

    \medskip

    \fbox{\box0}

    \setbox1=\vtopR{\hsize=3cm
        سطر نخست\\English\\سطر آخر\par}
    \fbox{\box1}
