\ifLuaTeX
\directlua{
  function latinverbatim_check(locale, family, starred)
    local letters, spaces = 0, 0
    local function walk(head)
      for n in node.traverse(head) do
        if n.id == node.id('glyph') then
          assert(font.getfont(n.font).characters[n.char], 'Missing glyph')
          if n.char == 65 then
            letters = letters + 1
            assert(node.get_attribute(n, Babel.attr_locale) == locale,
              'Wrong verbatim text locale: ' .. tostring(node.get_attribute(n, Babel.attr_locale)) .. ' expected ' .. locale)
            assert(string.find(font.getfont(n.font).fullname, family, 1, true),
              'Wrong verbatim font: ' .. font.getfont(n.font).fullname)
          elseif n.char == 32 or n.char == 0x2423 then
            spaces = spaces + 1
          end
        elseif n.id == node.id('hlist') or n.id == node.id('vlist') then
          walk(n.head)
        end
      end
    end
    walk(tex.box[0].head)
    assert(letters > 0, 'Missing literal test text')
    if starred then assert(spaces > 0, 'Starred form lost visible spaces') end
  end
}
\fi
