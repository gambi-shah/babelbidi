% Run from test with TEXINPUTS=..: and LuaLaTeX.
% Override \FallbackCache (true/false) and \FallbackOnchar (empty/fonts)
% on the command line to exercise all four mapper configurations.
\documentclass{article}
\providecommand\FallbackCache{true}
\providecommand\FallbackOnchar{}
\edef\fallbackoptions{fontcache=\FallbackCache}
\expandafter\usepackage\expandafter[\fallbackoptions]{babelbidi}
\babelprovide[main,import]{persian}
\ifx\FallbackOnchar\empty
  \babelprovide[import]{english}
\else
  \babelprovide[import,onchar=fonts]{english}
  \babelprovide[onchar=fonts]{persian}
\fi
\settextfont{HM XNiloofar}
\babelfont[english]{rm}{HM XNiloofar}
\setlatinmonofont{JetBrains Mono}
% Exercise both positions for fontspec's optional font features.
\setmonofont[Ligatures=TeX]{Vazir Code}[Scale=1]
\fontfallback{english}{persian}
\fontfallback{persian}{english}
\tracinglostchars=3

\directlua{
  function fallback_check_box(latin, persian)
    local seen_latin, seen_persian = false, false
    for n in node.traverse_id(node.id('glyph'), tex.box[0].head) do
      local f = font.getfont(n.font)
      assert(f.characters[n.char], 'Missing glyph after fallback')
      local expected
      if n.char == 65 then
        expected, seen_latin = latin, true
      elseif n.char > 127 then
        expected, seen_persian = persian, true
      end
      if expected then
        assert(string.find(f.fullname, expected, 1, true),
          'Expected ' .. expected .. ', got ' .. f.fullname)
      end
    end
    assert(seen_latin and seen_persian, 'Expected mixed Persian/Latin text')
  end
}

  \setbox0=\hbox{#1سلام A}%
