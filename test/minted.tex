% Save and verify the surrounding Babel and font context.
\makeatletter
\newcommand\SaveCodeContext{%
  \edef\SavedCodeContext{%
    \languagename/\the\textdir/\the\pardir/\f@family
  }%
  \let\SavedBabelAux\babel@aux
}

\newcommand\CheckCodeContext{%
  \edef\CurrentCodeContext{%
    \languagename/\the\textdir/\the\pardir/\f@family
  }%
  \ifx\CurrentCodeContext\SavedCodeContext\else
    \errmessage{Input minted leaked language, direction or font}%
  \fi
  \ifx\babel@aux\SavedBabelAux\else
    \errmessage{Input minted leaked its Babel aux override}%
  \fi
}
\makeatother
